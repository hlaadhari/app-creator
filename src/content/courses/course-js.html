4. Fonctions de Validation & Conversion de Types

JavaScript gère les valeurs des formulaires sous forme de chaînes de caractères (string). La conversion et les tests numériques sont indispensables.

Tests numériques et conversions JavaScript 
// isNaN() teste si une valeur N'EST PAS un nombre
isNaN("12345678"); // false (c'est bien un nombre valide)
isNaN("12A45");    // true (caractère invalide)
isNaN("");         // false (attention: chaîne vide évaluée à 0 !)

// Conversion de type
let note1 = parseInt("15");      // 15 (entier)
let note2 = parseFloat("14.75"); // 14.75 (décimal)
let total = Number("12.50");     // 12.5
let chaine = String(2024);       // "2024"

// Test rigoureux d'un entier positif
function estEntierPositif(val) {
  let n = Number(val);
  return !isNaN(val) && val.trim() !== "" && Number.isInteger(n) && n >= 0;
}

// Exemple : Calculer l'âge à partir d'une date de naissance (YYYY-MM-DD)
function calculerAge(dateNaissStr) {
  let dateN = new Date(dateNaissStr);
  let age = aujourdhui.getFullYear() - dateN.getFullYear();
  let m = aujourdhui.getMonth() - dateN.getMonth();
  if (m < 0 || (m === 0 && aujourdhui.getDate() < dateN.getDate())) {
    age--;
  }
  return age;
}

7. Patron Canonique de la Fonction verif() (Bac STI)

Voici l'implémentation type requise dans les épreuves pratiques, assurant la validation complète de l'ensemble des contrôles de saisie avant soumission au serveur PHP :

Fichier controle.js officiel JavaScript 
/**
 * Contrôle de saisie officiel - Épreuve Pratique Bac STI
 * Fichier : controle.js
 * Retourne false si une règle est violée, sinon true pour soumettre le formulaire.
 */
function verif() {
  // 1. Récupération des références du formulaire
  let fCin = document.getElementById("cin");
  let fNom = document.getElementById("nom");
  let fAge = document.getElementById("age");
  let rGenreM = document.getElementById("genreM");
  let rGenreF = document.getElementById("genreF");
  let sSection = document.getElementById("section");

  // 2. Contrôle du CIN : Exactement 8 chiffres obligatoires
  let cin = fCin.value.trim();
  if (cin.length !== 8 || isNaN(cin)) {
    alert("Le numéro de CIN doit être obligatoirement composé de 8 chiffres décimaux !");
    fCin.focus();
    return false;
  }

  // 3. Contrôle du Nom & Prénom : Non vide et uniquement alphabétique + espaces
  let nom = fNom.value.trim().toUpperCase();
  if (nom === "") {
    alert("Veuillez renseigner votre Nom et Prénom !");
    fNom.focus();
    return false;
  }
  for (let i = 0; i < nom.length; i++) {
    let car = nom.charAt(i);
    if ((car < "A" || car > "Z") && car !== " ") {
      alert("Le Nom et Prénom ne doivent comporter que des lettres alphabétiques !");
      fNom.focus();
      return false;
    }
  }

  // 4. Contrôle de l'Âge : Entier numérique entre 17 et 25
  let age = Number(fAge.value);
  if (isNaN(fAge.value) || fAge.value.trim() === "" || age < 17 || age > 25) {
    alert("L'âge doit être un entier valide compris entre 17 et 25 ans !");
    fAge.focus();
    return false;
  }

  // 5. Contrôle du Choix Radio (Genre) : Au moins un bouton coché
  if (!rGenreM.checked && !rGenreF.checked) {
    alert("Veuillez sélectionner votre genre (Masculin ou Féminin) !");
    return false;
  }

  // 6. Contrôle de la Liste Déroulante : Sélection d'une option valide (index > 0)
  if (sSection.selectedIndex === 0 || sSection.value === "") {
    alert("Veuillez choisir une section dans la liste déroulante !");
    sSection.focus();
    return false;
  }

  // Si toutes les validations ont réussi, autoriser la soumission
  return true;
}
